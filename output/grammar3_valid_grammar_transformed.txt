
--- ORIGINAL GRAMMAR ---
Stmt -> if Cond then Stmt else Stmt | if Cond then Stmt | assign
Cond -> id

--- AFTER LEFT FACTORING ---
Stmt -> assign | if Stmt'
Cond -> id
Stmt' -> Cond Stmt''
Stmt'' -> then Stmt'''
Stmt''' -> Stmt Stmt''''
Stmt'''' -> else Stmt | epsilon

--- AFTER LEFT RECURSION REMOVAL ---
Stmt -> assign | if Stmt'
Cond -> id
Stmt' -> id Stmt''
Stmt'' -> then Stmt'''
Stmt''' -> assign Stmt'''' | if Stmt' Stmt''''
Stmt'''' -> else Stmt | epsilon
